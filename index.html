Usage

Add the stylesheet, drop in the markup. Everything reads currentColor.

<!-- CDN -->
<link rel="stylesheet" href="https://unpkg.com/hmm.css">

<!-- pure CSS -->
<span class="hmm-shimmer">Thinking</span>
<span class="hmm-dots"><i></i><i></i><i></i></span>
<p>Streaming answer<span class="hmm-caret"></span></p>
// JS-powered ones (wave, glimmer, rise, roll, scramble)
import "hmm.css";
import "hmm.css/auto";   // wires up data-hmm-* attributes

// or drive them yourself
import { split, scramble, cycle } from "hmm.css/js";
const stop = cycle(el, ["Thinking", "Searching", "Reading"]);
await scramble(el, "Found 12 sources");
--hmm-durationLength of one cycle. Each indicator has its own default.
--hmm-sizeSize of shape indicators: dots, orb, aura, sparkle and the rest.
--hmm-highlightColour of the shimmer sweep. Defaults to a lighter currentColor.
--hmm-a … --hmm-dPalette for the multi-colour indicators: aura, sparkle, blob and trace.
--hmm-radius, --hmm-widthCorner radius and stroke width for .hmm-trace.
